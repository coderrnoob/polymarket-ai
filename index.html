import { useState, useEffect, useRef, useCallback } from "react"; // ── INDICATORS ───────────────────────────────────────────── function calcRSI(closes, period = 14) { if (closes.length < period + 1) return 50; let gains = 0, losses = 0; for (let i = closes.length - period; i < closes.length; i++) { const d = closes[i] - closes[i - 1]; if (d > 0) gains += d; else losses += Math.abs(d); } const ag = gains / period, al = losses / period; if (al === 0) return 100; return 100 - 100 / (1 + ag / al); } function calcEMA(arr, period) { const k = 2 / (period + 1); let ema = arr.slice(0, period).reduce((a, b) => a + b, 0) / period; for (let i = period; i < arr.length; i++) ema = arr[i] * k + ema * (1 - k); return ema; } function calcMACDSeries(closes) { const k12 = 2/13, k26 = 2/27, k9 = 2/10; let e12 = closes[0], e26 = closes[0]; const hists = []; for (let i = 1; i < closes.length; i++) { e12 = closes[i]*k12 + e12*(1-k12); e26 = closes[i]*k26 + e26*(1-k26); hists.push(e12 - e26); } let sig = hists[0]; return hists.map(h => { sig = h*k9 + sig*(1-k9); return { macd:h, signal:sig, hist:h-sig }; }); } function fmtPrice(n) { return Number(n).toLocaleString('en-US',{minimumFractionDigits:2,maximumFractionDigits:2}); } function fmtVol(v) { v = parseFloat(v); if (v >= 1e9) return (v/1e9).toFixed(2)+'B'; if (v >= 1e6) return (v/1e6).toFixed(2)+'M'; if (v >= 1e3) return (v/1e3).toFixed(2)+'K'; return v.toFixed(2); } function computePrediction({ rsiCls, volCls, macdCls, emaCls, rsi, candles }) { const s = { bull:0, bear:0, neutral:0 }; const reasons = []; if (rsiCls==='bull') { s.bull+=25; reasons.push('RSI oversold (<30) — reversal likely UP'); } else if (rsiCls==='bear') { s.bear+=25; reasons.push('RSI overbought (>70) — reversal likely DOWN'); } else { s.neutral+=10; reasons.push(`RSI at ${rsi.toFixed(1)} — no extreme reading`); } if (macdCls==='bull') { s.bull+=25; reasons.push('MACD bullish — momentum shifting upward'); } else if (macdCls==='bear') { s.bear+=25; reasons.push('MACD bearish — selling pressure rising'); } else { s.neutral+=10; } if (volCls==='bull') { s.bull+=20; reasons.push('Volume spike on green candles — buyers active'); } else if (volCls==='bear') { s.bear+=20; reasons.push('Volume spike on red candles — sellers dominant'); } else { s.neutral+=10; reasons.push('Volume near average — no directional conviction'); } if (emaCls==='bull') { s.bull+=20; reasons.push('Price above EMA 9/21/50 — uptrend intact'); } else if (emaCls==='bear') { s.bear+=20; reasons.push('Price below EMA 9/21/50 — downtrend active'); } else { s.neutral+=10; reasons.push('Price mixed across EMAs — trend unclear'); } const last3 = candles.slice(-4,-1); const consecGreen = last3.every(c => c.close >= c.open); const consecRed = last3.every(c => c.close < c.open); let paCls = 'neutral'; if (consecGreen) { s.bull+=10; paCls='bull'; reasons.push('3 consecutive green candles — bullish momentum'); } else if (consecRed) { s.bear+=10; paCls='bear'; reasons.push('3 consecutive red candles — bearish momentum'); } else { s.neutral+=5; } const total = s.bull + s.bear + s.neutral || 1; let cls, conf; if (s.bull >= s.bear && s.bull >= s.neutral) { cls='bull'; conf=Math.round((s.bull/total)*100); } else if (s.bear > s.bull && s.bear >= s.neutral) { cls='bear'; conf=Math.round((s.bear/total)*100); } else { cls='neutral'; conf=Math.round((s.neutral/total)*100); } return { cls, conf, reasons, paCls }; } // ── GENERATE REALISTIC CANDLES VIA CLAUDE API ────────────── async function fetchCandlesFromClaude(tf) { const tfMinutes = tf==='5m'?5:tf==='15m'?15:tf==='1h'?60:240;
